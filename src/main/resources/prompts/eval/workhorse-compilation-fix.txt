    @Query("SELECT i FROM Invoice i WHERE i.status = :status AND i.createdAt < :before")

@Service
class BillingService {

    private final InvoiceRepository invoiceRepository;
    private final Map<String, BigDecimal> customerBalanceCache = new HashMap<>();

    public BillingService(InvoiceRepository invoiceRepository) {
        this.invoiceRepository = invoiceRepository;
    }

    @Transactional
    public Invoice createInvoice(String customerId, List<LineItem> lineItems) {
        Invoice invoice = new Invoice();
        invoice.setCustomerId(customerId);
        invoice.setCreatedAt(LocalDateTime.now());

        BigDecimal total = lineItems.stream()
                .map(li -> li.unitPrice.multiply(BigDecimal.valueOf(li.quantity)))
                .reduce(BigDecimal::add)
                .get();

        invoice.setTotalAmount(total);
        return invoiceRepository.save(invoice);
    }

    @Transactional
    public Invoice applyPayment(String invoiceId, BigDecimal amount) {
        Invoice invoice = invoiceRepository.findById(invoiceId).orElseThrow();

        if (invoice.getStatus() == Invoice.InvoiceStatus.CANCELLED) {
            throw new IllegalStateException("Cannot pay cancelled invoice");
        }

        BigDecimal newPaid = invoice.getPaidAmount().add(amount);
        invoice.setPaidAmount(newPaid);

        if (newPaid.compareTo(invoice.getTotalAmount()) >= 0) {
            invoice.setStatus(Invoice.InvoiceStatus.PAID);
            invoice.setPaidAt(LocalDateTime.now());
        } else if (newPaid.compareTo(BigDecimal.ZERO) > 0) {
            invoice.setStatus(Invoice.InvoiceStatus.PARTIALLY_PAID);
        }

        // Update balance cache
        BigDecimal remaining = invoice.getTotalAmount().subtract(newPaid);
        customerBalanceCache.put(invoice.getCustomerId(), remaining);

        return invoiceRepository.save(invoice);
    }

    public BigDecimal getCustomerBalance(String customerId) {
        if (customerBalanceCache.containsKey(customerId)) {
            return customerBalanceCache.get(customerId);
        }

        List<Invoice> invoices = invoiceRepository.findByCustomerId(customerId);
        BigDecimal balance = invoices.stream()
                .filter(i -> i.getStatus() != Invoice.InvoiceStatus.CANCELLED)
                .map(i -> i.getTotalAmount().subtract(i.getPaidAmount()))
                .reduce(BigDecimal::add)
                .get();

        customerBalanceCache.put(customerId, balance);
        return balance;
    }

    @Transactional
    public void markOverdueInvoices() {
        LocalDateTime thirtyDaysAgo = LocalDateTime.now().minusDays(30);
        List<Invoice> overdue = invoiceRepository.findOverdueInvoices(
                Invoice.InvoiceStatus.PENDING, thirtyDaysAgo);

        overdue.forEach(invoice -> {
            invoice.setStatus(Invoice.InvoiceStatus.OVERDUE);
        });
        invoiceRepository.saveAll(overdue);
    }

    public Map<String, Object> generateReport(String customerId) {
        List<Invoice> invoices = invoiceRepository.findByCustomerId(customerId);

        Map<Invoice.InvoiceStatus, Long> statusCounts = invoices.stream()
                .collect(Collectors.groupingBy(Invoice::getStatus, Collectors.counting()));

        double avgPaymentTime = invoices.stream()
                .filter(i -> i.getPaidAt() != null)
                .mapToLong(i -> java.time.Duration.between(i.getCreatedAt(), i.getPaidAt()).toDays())
                .average()
                .getAsDouble();

        Map<String, Object> report = new HashMap<>();
        report.put("total_invoices", invoices.size());
        report.put("status_breakdown", statusCounts);
        report.put("avg_payment_days", avgPaymentTime);
        report.put("outstanding_balance", getCustomerBalance(customerId));
        return report;
    }
